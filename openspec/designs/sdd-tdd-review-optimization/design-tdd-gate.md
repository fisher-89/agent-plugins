```python
def detect_test_framework(project_root: str) -> tuple[str, str]:
    """
    Returns: (framework, runner_command)
    """
    package_json = os.path.join(project_root, "package.json")
    requirements_txt = os.path.join(project_root, "requirements.txt")
    pyproject_toml = os.path.join(project_root, "pyproject.toml")

    if os.path.exists(package_json):
        # Node.js project
        with open(package_json) as f:
            pkg = json.load(f)
            deps = {**pkg.get("dependencies", {}),
                    **pkg.get("devDependencies", {})}

            if "vitest" in deps:
                return ("vitest", "npx vitest run")
            if "jest" in deps:
                return ("jest", "npm test")
            if "mocha" in deps:
                return ("mocha", "npx mocha")
            # Default to jest
            return ("jest", "npm test")

    if os.path.exists(pyproject_toml):
        # Python project with pyproject.toml
        return ("pytest", "pytest")

    if os.path.exists(requirements_txt):
        # Python project with requirements.txt
        with open(requirements_txt) as f:
            content = f.read()
            if "pytest" in content:
                return ("pytest", "pytest")

    # Fallback
    return ("unknown", "")
```

#### 2.2.2 影响文件检测

**Heuristic 策略**:

```python
def infer_affected_files(task: str, src_dir: str) -> list[str]:
    """Infer affected source files from task description."""

    # Strategy 1: Explicit file mentioned
    # Task: "Implement user registration in auth.js"
    file_match = re.search(r'(?:in|modify|edit|update)\s+([a-zA-Z_]+\.[a-z]+)', task)
    if file_match:
        filename = file_match.group(1)
        return [os.path.join(src_dir, filename)]

    # Strategy 2: Feature module inference
    # Task: "Add JWT authentication"
    keywords = extract_keywords(task)
    for kw in keywords:
        possible_file = os.path.join(src_dir, f"{kw}.js")
        if os.path.exists(possible_file):
            return [possible_file]

    # Strategy 3: Glob search
    # Task: "Implement routes for user management"
    if "routes" in task.lower():
        return glob.glob(os.path.join(src_dir, "routes", "*.js"))

    return []
```

```markdown
### Test Failure Handling

When tests fail:

   - [2] Mark as blocked, explain issue
   - [3] Skip test validation (requires user confirmation)
4. Do NOT mark task complete until tests pass
```

#### 2.3.2 检测逻辑

```python
def check_test_coverage(task: str, changes: list[str], tests_dir: str) -> dict:
    """
    Check if changes have corresponding tests.

    Returns:
        {
            "has_tests": bool,
            "missing_tests": list[str],
            "existing_tests": list[str]
        }
    """
    missing_tests = []
    existing_tests = []

    for changed_file in changes:
        # Map source file to test file
        test_file = map_to_test_file(changed_file, tests_dir)

        if os.path.exists(test_file):
            existing_tests.append(test_file)
        else:
            missing_tests.append(test_file)

    return {
        "has_tests": len(missing_tests) == 0,
        "missing_tests": missing_tests,
        "existing_tests": existing_tests
    }

def map_to_test_file(source_file: str, tests_dir: str) -> str:
    """Map source file to expected test file path."""
    # src/routes/auth.js → tests/routes/auth.test.js
    # src/auth.py → tests/test_auth.py

    basename = os.path.basename(source_file)
    name, ext = os.path.splitext(basename)

    if ext in (".js", ".ts", ".jsx", ".tsx"):
        return os.path.join(tests_dir, f"{name}.test{ext}")
    elif ext == ".py":
        return os.path.join(tests_dir, f"test_{name}.py")

    return os.path.join(tests_dir, f"test_{name}.{ext}")
```

## 4. 兼容性考虑

### 4.1 项目无测试框架

**处理**:
1. 检测不到测试框架时，fallback 到 markdown 模板
2. 提示用户安装测试框架
3. 提供 `npm install --save-dev jest` 或 `pip install pytest` 命令

### 4.2 测试框架自定义配置

**处理**:
1. 读取 jest.config.js / pytest.ini / vitest.config.ts
2. 提取 testMatch / testPathIgnorePatterns
3. 按配置生成测试文件路径

### 4.3 跳过测试的合理场景

允许用户主动跳过测试验证：
- 纯配置文件修改 (.gitignore, README.md)
- 样式文件修改 (.css, .scss)
- 文档文件修改 (.md)

## 5. 风险与缓解

| 风险 | 影响 | 缓解措施 |
|------|------|----------|
| 测试框架检测错误 | 生成错误模板 | 多维度检测，fallback 机制 |
| 影响文件推断不准 | 测试覆盖率检查遗漏 | 启发式 + 用户确认 |
| 测试运行超时 | 流程卡死 | 设置 60s 超时，跳过并警告 |
| 强制测试降低效率 | 开发体验变差 | 允许用户确认跳过 |

```python
def detect_linter(project_root: str) -> tuple[str, str]:
    """Detect linter/type checker in use."""
    # Node.js
    package_json = os.path.join(project_root, "package.json")
    if os.path.isfile(package_json):
        pkg = json.load(open(package_json))
        deps = {**pkg.get("devDependencies", {})}
        if "eslint" in deps:
            return ("eslint", "npx eslint")
        if "biome" in deps:
            return ("biome", "npx biome check")

    # Python
    pyproject = os.path.join(project_root, "pyproject.toml")
    if os.path.isfile(pyproject):
        return ("ruff", "ruff check")

    return ("unknown", "")
```
